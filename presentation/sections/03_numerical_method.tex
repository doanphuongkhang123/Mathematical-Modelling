\begin{frame}{Euler and RK4}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.42\textwidth}
      \keypoint{Forward Euler}
      \[
        X_{n+1}=X_n+hF(X_n)
      \]
      \begin{itemize}
        \item Transparent first-order baseline
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \keypoint{Classical Runge--Kutta}
      \begin{align*}
        k_1 &= F(X_n),\\
        k_2 &= F(X_n+\tfrac h2k_1),\\
        k_3 &= F(X_n+\tfrac h2k_2),\\
        k_4 &= F(X_n+hk_3),\\[-0.05cm]
        X_{n+1} &= X_n+\frac h6(k_1+2k_2+2k_3+k_4).
      \end{align*}
      \begin{itemize}
        \item Fourth-order method for long, oscillatory runs
        \item Partial step reaches $t_{\mathrm{end}}$ exactly
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
