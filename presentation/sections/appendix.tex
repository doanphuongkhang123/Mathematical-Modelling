\begin{frame}[t]{Backup: Equilibrium Formulas}
  \fontsize{10}{12}\selectfont
  \begin{align*}
    P_0 &= (1,0,0,0),\\
    P_1 &= (S_1,x_1,0,0),\quad
    S_1=f_1^{-1}(D_1),\quad
    x_1=\frac{1-S_1}{D_1},\\
    P_2 &= (S_2,x_2,y_2,0),\quad
    x_2=f_2^{-1}(D_2),\\
    S_2+f_1(S_2)x_2&=1,\qquad
    y_2=\frac{x_2(f_1(S_2)-D_1)}{D_2},\\
    P_3 &= (S_3,x_3,y_3,z_3),\quad
    y_3=f_3^{-1}(D_3),\\
    S_3+f_1(S_3)x_3&=1,\qquad
    x_3(f_1(S_3)-D_1)=f_2(x_3)y_3,\\
    z_3&=\frac{y_3(f_2(x_3)-D_2)}{D_3}.
  \end{align*}
\end{frame}

\begin{frame}[t]{Backup: Jacobian and Routh--Hurwitz}
  \small
  \[
  J=\begin{pmatrix}
  -1-f_1'(S)x & -f_1(S) & 0 & 0\\
  f_1'(S)x & f_1(S)-D_1-f_2'(x)y & -f_2(x) & 0\\
  0 & f_2'(x)y & f_2(x)-D_2-f_3'(y)z & -f_3(y)\\
  0 & 0 & f_3'(y)z & f_3(y)-D_3
  \end{pmatrix}.
  \]

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.47\textwidth}
      \keypoint{$P_2$ cubic}
      \[
        \lambda^3+A_1\lambda^2+A_2\lambda+A_3
      \]
      \[
        A_1,A_2,A_3>0,\qquad A_1A_2>A_3,
      \]
      together with $f_3(y_2)-D_3<0$.
    \end{column}
    \begin{column}{0.49\textwidth}
      \keypoint{$P_3$ quartic}
      \[
        \lambda^4+C_1\lambda^3+C_2\lambda^2+C_3\lambda+C_4
      \]
      \[
        C_i>0,\quad C_1C_2>C_3,
      \]
      \[
        C_1C_2C_3>C_3^2+C_1^2C_4.
      \]
    \end{column}
  \end{columns}
\end{frame}
